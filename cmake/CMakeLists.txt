################################################################################
# Provisioning (#15): the bridge's serial and EUI-64, and the trust anchor
# for the server certificate. Device values must not enter git
# (CLAUDE.md), so they come from config/local.cmake, which is gitignored;
# template: config/local.cmake.example. Written to
# <build>/generated/provision.h. Without them the firmware builds but does
# not connect to the server. P6 moves serial and EUI-64 into the identity
# record in external flash (docs/design.md "Persistent storage").
set(LOCAL_CONFIG "${REPO_ROOT}/config/local.cmake")
if(EXISTS ${LOCAL_CONFIG})
    include(${LOCAL_CONFIG})
    message("Provisioning: config/local.cmake")
else()
    message("Provisioning: no config/local.cmake; the firmware will not connect to the server")
endif()
set(PROVISION "/* Generated by cmake/CMakeLists.txt from config/local.cmake. Do not commit. */\n")
string(APPEND PROVISION "#ifndef PROVISION_H\n#define PROVISION_H\n")
if(DEFINED BRIDGE_SERIAL)
    if(NOT BRIDGE_SERIAL MATCHES "^[A-Z][A-Z][0-9A-Z]+$")
        message(FATAL_ERROR "BRIDGE_SERIAL '${BRIDGE_SERIAL}' is not of the form IB0123456789")
    endif()
    string(APPEND PROVISION "#define PROVISION_SERIAL \"${BRIDGE_SERIAL}\"\n")
endif()
if(DEFINED BRIDGE_EUI64)
    string(TOLOWER "${BRIDGE_EUI64}" eui)
    string(REGEX REPLACE "[: -]" "" eui "${eui}")
    if(NOT eui MATCHES "^[0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f]$")
        message(FATAL_ERROR "BRIDGE_EUI64 must be 8 bytes in hex, as stored (00 1b c5 ...)")
    endif()
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1, " eui "${eui}")
    string(APPEND PROVISION "#define PROVISION_EUI64 {${eui}}\n")
endif()
if(DEFINED SERVER_CA_PEM)
    if(NOT EXISTS ${SERVER_CA_PEM})
        message(FATAL_ERROR "SERVER_CA_PEM ${SERVER_CA_PEM} not found")
    endif()
    file(READ ${SERVER_CA_PEM} pem)
    if(pem MATCHES "PRIVATE KEY" OR NOT pem MATCHES "-----BEGIN CERTIFICATE-----")
        message(FATAL_ERROR "SERVER_CA_PEM must hold certificates only, never a key")
    endif()
    string(REPLACE "\r" "" pem "${pem}")
    string(STRIP "${pem}" pem)
    string(REPLACE "\n" "\\n\" \\\n    \"" pem "${pem}")
    string(APPEND PROVISION "#define PROVISION_CA_PEM \\\n    \"${pem}\\n\"\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${SERVER_CA_PEM})
endif()
string(APPEND PROVISION "#endif\n")
# Rewritten only when the content changes, so nothing rebuilds needlessly.
file(WRITE ${CMAKE_BINARY_DIR}/generated/provision.h.tmp "${PROVISION}")
configure_file(${CMAKE_BINARY_DIR}/generated/provision.h.tmp
               ${CMAKE_BINARY_DIR}/generated/provision.h COPYONLY)

        -Wno-volatile

        -Wuseless-cast
